# 二、SQL 建表（核心可用版）

以下是**精簡但完整可上線版本（MySQL/PostgreSQL皆可微調）**

---

## 1️⃣ 員工與職位

CREATE TABLE employees (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    name VARCHAR(50),  
    phone VARCHAR(20),  
    status VARCHAR(20),  
    hire_date DATE  
);  
  
CREATE TABLE positions (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    name VARCHAR(50)  
);  
  
CREATE TABLE employee_positions (  
    employee_id BIGINT,  
    position_id BIGINT,  
    PRIMARY KEY (employee_id, position_id)  
);

---

## 2️⃣ 案場與崗位

CREATE TABLE sites (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    name VARCHAR(100),  
    status VARCHAR(20)  
);  
  
CREATE TABLE site_posts (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    site_id BIGINT,  
    position_id BIGINT,  
    name VARCHAR(50)  
);

---

## 3️⃣ 班別與需求

CREATE TABLE shifts (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    name VARCHAR(50),  
    start_time TIME,  
    end_time TIME  
);  
  
CREATE TABLE post_shifts (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    post_id BIGINT,  
    shift_id BIGINT,  
    required_count INT  
);

---

## 4️⃣ 排班（核心）

CREATE TABLE schedule_batches (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    month VARCHAR(7),  
    status VARCHAR(20)  
);  
  
CREATE TABLE schedules (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    batch_id BIGINT,  
    employee_id BIGINT,  
    post_shift_id BIGINT,  
    work_date DATE,  
    work_type VARCHAR(20), -- monthly / cash_daily / cash_hourly  
    status VARCHAR(20),  
    UNIQUE(employee_id, work_date)  
);

---

## 5️⃣ 出勤

CREATE TABLE attendance (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    schedule_id BIGINT,  
    check_in DATETIME,  
    check_out DATETIME,  
    work_hours DECIMAL(5,2)  
);

---

## 6️⃣ 薪資規則（超核心）

CREATE TABLE pay_rates (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    employee_id BIGINT NULL,  
    post_id BIGINT NULL,  
    shift_id BIGINT NULL,  
    work_type VARCHAR(20),  
    pay_type VARCHAR(20), -- hour / day  
    rate DECIMAL(10,2),  
    priority INT  
);

---

## 7️⃣ 薪資紀錄

CREATE TABLE payroll_items (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    employee_id BIGINT,  
    schedule_id BIGINT,  
    amount DECIMAL(10,2),  
    pay_type VARCHAR(20), -- monthly / cash  
    status VARCHAR(20)  
);

---

## 8️⃣ 現金班結算

CREATE TABLE cash_pay_batches (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    employee_id BIGINT,  
    start_date DATE,  
    end_date DATE,  
    total_amount DECIMAL(10,2),  
    paid_date DATE  
);

---

## 9️⃣ 月薪

CREATE TABLE employee_salary (  
    employee_id BIGINT PRIMARY KEY,  
    base_salary DECIMAL(10,2),  
    labor_insurance DECIMAL(10,2),  
    health_insurance DECIMAL(10,2),  
    pension DECIMAL(10,2)  
);

---

## 🔟 排班限制

CREATE TABLE employee_schedule_rules (  
    employee_id BIGINT PRIMARY KEY,  
    max_consecutive_night INT,  
    max_consecutive_work_days INT  
);  
  
CREATE TABLE employee_unavailable_dates (  
    id BIGINT PRIMARY KEY AUTO_INCREMENT,  
    employee_id BIGINT,  
    date DATE  
);

---

# 三、核心流程圖（排班）

---

## 🧠 排班流程（半自動）

開始  
 ↓  
讀取案場需求（post_shifts）  
 ↓  
讀取可用員工（排除請假）  
 ↓  
依班別排序（先夜班）  
 ↓  
for 每一天  
  for 每個崗位班別  
    篩選可用員工  
    過濾限制（夜班/連續上班）  
    選「工時最少」的人  
    寫入 schedules  
 ↓  
檢查缺工  
 ↓  
人工調整  
 ↓  
完成

---

# 四、薪資計算流程

---

## 💰 現金班

讀取 schedules（cash）  
 ↓  
找 pay_rate（優先級匹配）  
 ↓  
計算金額  
 ↓  
寫入 payroll_items  
 ↓  
依週期分組  
 ↓  
產生 cash_pay_batches

---

## 💰 月薪

讀取 employee_salary  
 ↓  
計算出勤（缺勤/加班）  
 ↓  
加總 cash 班收入  
 ↓  
扣保險  
 ↓  
產生 monthly_payroll

---
